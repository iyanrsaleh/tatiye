App.js
// App.js — entry runtime NXUI.Tatiye
import "./assets/modules/nxdom.js";
import config from "./config.js";

new NXUI.Tatiye({
  endpoint: config,
  containerId: "main",
  appRoot: "templates",
  route: ["beranda", "autoload"],
}).run();

// templates/beranda.js — sebuah halaman
export async function beranda(page, route) {
  route.register(page, async (name, container) => {
    container.innerHTML = `<h1>Nxdom Framework</h1>`;
  });
}

Tentang Kami

Perusahaan teknologi digital yang membangun produk perangkat lunak untuk komunitas global.

PT Tatiye Digital Solusindo adalah perusahaan induk di balik pengembangan framework tatiye — library JavaScript untuk antarmuka DOM — serta produk turunan yang tumbuh menjadi ekosistem NXDOM.

Kami percaya perangkat lunak yang baik lahir dari fondasi yang kokoh: arsitektur yang rapi, dokumentasi yang terbuka, dan komitmen jangka panjang. Karena itu kami membangun framework, SDK, dan tools yang siap dipelajari serta digunakan bersama.

  • ◆
    Framework open source

    NXUI.Tatiye — runtime SPA/PWA berbasis ES6.

  • ⬡
    Ekosistem produk

    Tujuh produk di bawah payung NXDOM, dari CLI hingga database.

  • ⚙
    Rekayasa perangkat lunak

    Pengembangan web, desktop, SDK, dan infrastruktur data.
